// Original path: apps/desktop/src/main/zz-security-packaging-1-renderer-loads-from-file-and-the-grantfi.test.ts -- result: FAILS as expected (grantFileProtocolExtraPrivileges is undefined, i.e. Electron default on); sanity test passes
import { readFileSync } from "node:fs";
import { fileURLToPath } from "node:url";
import { describe, expect, it } from "vitest";

const config = readFileSync(
  fileURLToPath(new URL("../../electron-builder.yml", import.meta.url)),
  "utf8",
);

/** The electronFuses block of electron-builder.yml, as key: value pairs. */
function fuses(): Record<string, string> {
  const block = /^electronFuses:\n((?:[ \t]+.*\n|[ \t]*#.*\n|\n)*)/m.exec(config)?.[1] ?? "";
  const entries: Record<string, string> = {};
  for (const line of block.split("\n")) {
    const match = /^\s+([A-Za-z]+):\s*(\S+)/.exec(line);
    if (match?.[1] !== undefined && match[2] !== undefined) entries[match[1]] = match[2];
  }
  return entries;
}

describe("packaged app fuses", () => {
  it("parses the fuse block", () => {
    expect(fuses().runAsNode).toBe("false");
  });

  it("turns off the extra file:// privileges (fetch of any file:// URL from the renderer)", () => {
    expect(fuses().grantFileProtocolExtraPrivileges).toBe("false");
  });
});

// Original path: apps/desktop/src/main/zz-security-publish-1-stored-publish-secrets-are-keyed-only-by.test.ts. Result: fails as expected (expected 'hunter2' to be null).
import { join } from "node:path";
import { tempDirFactory } from "@webforge/test-utils/vitest";
import { describe, expect, it } from "vitest";
import { createCredentialStore, credentialKey, type SecretCipher } from "./credential-store.js";

const cipher: SecretCipher = {
  isEncryptionAvailable: () => true,
  encryptString: (text) => Buffer.from(`enc:${text}`),
  decryptString: (data) => data.toString().slice(4),
};

// The `.wforge` file is shareable; every field below comes from it.
type FileDestination = { projectId: string; profileId: string; protocol: "ftp" | "ftps"; host: string };

describe("security: stored publish secret is bound to its destination", () => {
  const tempDir = tempDirFactory();

  it("does not hand the real host's password to a destination whose host was edited in the file", async () => {
    const store = createCredentialStore(join(tempDir(), "credentials.json"), cipher);
    const original: FileDestination = {
      projectId: "7d2b0c1e-1111-4222-8333-444455556666",
      profileId: "live",
      protocol: "ftps",
      host: "ftp.real.example",
    };
    // The user saves their hosting password for the destination they typed.
    await store.set(credentialKey(original.projectId, original.profileId), "hunter2");

    // A recipient edits the shared file: same IDs, attacker's host, plain FTP.
    const tampered: FileDestination = { ...original, protocol: "ftp", host: "evil.example" };

    // The only lookup the store offers takes the IDs from the file, nothing about the host.
    const secret = await store.get(credentialKey(tampered.projectId, tampered.profileId));
    expect(secret).toBeNull();
  });
});
